10 REM *** PC-1600 BMP280: TEMPERATURE, PRESSURE, I2C SPEED ***
20 REM NEEDS Z80 I2C V1.0 AT &C400. SCL=PIN 29 SDA=PINS 27+31. BMP280 AT &77

300 REM --- SETUP: OVERSAMPLING AND IIR FILTER (1,2,4,8,16; ENTER = 1) ---
310 WAIT 128 : PRINT "1,2,4,8,16 ENTER=1" : WAIT 0
320 X = 1 : INPUT "temp oversampling ? "; X : GOSUB 800 : IF C < 0 THEN 320

400 REM --- ONE MEASUREMENT: FORCED MODE (MODE = 01) ---
410 POKE TX, &F4, CT * 32 + CP * 4 + 1 : POKE IA, D, 2, 0 : CALL IC, E
420 IF E <> 0 THEN 990
430 K = 0
440 GOSUB 700 : K = K + 1 : IF (S AND 8) <> 0 AND K < 200 THEN 440
445 IF K >= 200 THEN PRINT "BMP280 TIMEOUT" : END
450 POKE TX, &F7 : POKE IA, D, 1, 6 : CALL IC, E : IF E <> 0 THEN 990
460 UP = PEEK (RX) * 4096 + PEEK (RX + 1) * 16 + INT (PEEK (RX + 2) / 16)
470 UT = PEEK (RX + 3) * 4096 + PEEK (RX + 4) * 16 + INT (PEEK (RX + 5) / 16)
480 REM --- COMPENSATION (DATASHEET, FLOATING POINT) ---
485 V1 = (UT / 16384 - T1 / 1024) * T2
490 V2 = UT / 131072 - T1 / 8192 : V2 = V2 * V2 * T3
495 TF = V1 + V2 : TC = TF / 5120
515 V1 = TF / 2 - 64000 : V2 = V1 * V1 * P6 / 32768
520 V2 = V2 + V1 * P5 * 2 : V2 = V2 / 4 + P4 * 65536
525 V1 = (P3 * V1 * V1 / 524288 + P2 * V1) / 524288
530 V1 = (1 + V1 / 32768) * P1 : PA = 0 : IF V1 = 0 THEN 600
535 PA = 1048576 - UP : PA = (PA - V2 / 4096) * 6250 / V1
540 V1 = P9 * PA * PA / 2147483648 : V2 = PA * P8 / 32768
545 PA = PA + (V1 + V2 + P7) / 16
600 REM --- DISPLAY: T, P, SPEED; S = MEASURE SPEED AGAIN, OTHER KEY STOPS ---
610 Y = TC : GOSUB 850 : O$ = "Temperature = " + Q$ + " C"
620 WAIT 128 : PRINT O$
630 Y = PA / 100 : GOSUB 850 : O$ = "Pressure = " + Q$ + " hPa"
640 PRINT O$
650 PRINT "I2C approx. "; FQ; " kHz"
660 K$ = INKEY$ : IF K$ = "" THEN 410
670 IF K$ = "S" OR K$ = "s" THEN GOSUB 1000 : GOTO 410
680 WAIT 0 : PRINT "STOPPED" : END
700 REM --- STATUS REGISTER &F3 INTO S ---
710 POKE TX, &F3 : POKE IA, D, 1, 1 : CALL IC, E : IF E <> 0 THEN 990
720 S = PEEK (RX) : RETURN
800 REM --- X = 1,2,4,8,16 INTO C = 0..4 (X = 2^C), -1 = WRONG ---
810 C = -1 : V = 1 : FOR J = 0 TO 4 : IF X = V THEN LET C = J
820 V = V * 2 : NEXT J : RETURN
850 REM --- Y AS TEXT WITH 2 DECIMALS (-NN.NN) INTO Q$ ---
852 Z = Y : IF Z < 0 THEN LET Z = -Z
853 N = INT (Z * 100 + .5) : M = N : Q$ = ""
854 R = N - 10 * INT (N / 10) : N = INT (N / 10)
855 Q$ = CHR$ (48 + R) + Q$
856 IF LEN (Q$) = 2 THEN LET Q$ = "." + Q$
858 IF N > 0 OR LEN (Q$) < 4 THEN 854
860 IF Y < 0 AND M > 0 THEN LET Q$ = "-" + Q$
862 RETURN
900 REM --- 16-BIT VALUE FROM RX+N (LSB FIRST) INTO X ---
905 X = PEEK (RX + N) + 256 * PEEK (RX + N + 1) : RETURN
910 GOSUB 900 : IF X > 32767 THEN LET X = X - 65536
915 RETURN
980 WAIT 0 : PRINT "BAD CHIP ID "; I : END
990 WAIT 0 : PRINT "I2C ERROR E="; E : BEEP 3 : END
1000 REM --- I2C SPEED: 255-BYTE READS FOR 4 S (CLOCK = TIME), RESULT FQ ---
1010 WAIT 64 : PRINT "Measuring I2C speed..." : WAIT 0
1020 POKE TX, &88 : POKE IA, D, 1, 255
1030 GOSUB 1100 : Y = X
1040 GOSUB 1100 : IF X = Y THEN 1040
1050 W = X : N = 0
1060 FOR J = 1 TO 4 : CALL IC, E : NEXT J : IF E <> 0 THEN 990
1065 N = N + 4 : GOSUB 1100 : H = X - W : IF H < 0 THEN LET H = H + 3600
1070 IF H < 4 THEN 1060
1080 REM 2322 SCL CLOCKS PER READ: ADDR+REG 18, ADDR 9, 255 BYTES * 9
1090 FQ = INT (N * 2322 / H / 1000 + .5) : MS = INT (H * 1000 / N + .5)
1095 PRINT "255 B READ = "; MS; " ms" : RETURN
1100 REM --- SECONDS OF THE HOUR FROM TIME (MMDDHH.mmss) INTO X ---
1110 Q = INT ((TIME - INT (TIME)) * 10000 + .5)
1120 X = INT (Q / 100) * 60 + Q - 100 * INT (Q / 100) : RETURN
